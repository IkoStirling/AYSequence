cmake_minimum_required(VERSION 3.25)
project(AYSequence VERSION 0.1.0 LANGUAGES CXX)

add_library(AYSequence STATIC src/Sequence.cpp src/LightShow.cpp src/LightShowFog.cpp)
add_library(AYSequence::Core ALIAS AYSequence)
target_compile_features(AYSequence PUBLIC cxx_std_23)
if(NOT TARGET AYAssetFormat)
    add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../AYFoundation/AYAssetFormat
        ${CMAKE_CURRENT_BINARY_DIR}/AYAssetFormat EXCLUDE_FROM_ALL)
endif()
target_link_libraries(AYSequence PRIVATE AYAssetFormat)
target_include_directories(AYSequence PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
)
add_library(AYSequenceEditorCore STATIC src/SequenceAuthoring.cpp src/LightShowAuthoring.cpp src/LightShowFogAuthoring.cpp)
add_library(AYSequence::EditorCore ALIAS AYSequenceEditorCore)
target_link_libraries(AYSequenceEditorCore PUBLIC AYSequence)

option(AY_SEQUENCE_BUILD_ASSET_IO "Build sequence JSON/file support (requires AYIO)" ${AY_BUILD_RUNTIME})
if(AY_SEQUENCE_BUILD_ASSET_IO)
    find_package(nlohmann_json CONFIG REQUIRED)
    if(NOT TARGET AYIO)
        message(FATAL_ERROR "AYSequenceAssetIO requires AYIO; disable AY_SEQUENCE_BUILD_ASSET_IO for standalone Core")
    endif()
    add_library(AYSequenceAssetIO STATIC src/SequenceAssetIO.cpp src/SequenceDependencies.cpp src/SequenceCooked.cpp src/LightShowAssetIO.cpp src/LightShowBundle.cpp)
    add_library(AYSequence::AssetIO ALIAS AYSequenceAssetIO)
    target_link_libraries(AYSequenceAssetIO PUBLIC AYSequence AYIO PRIVATE AYAssetFormat nlohmann_json::nlohmann_json)
    add_executable(AYSequenceCookTool tools/SequenceCookTool.cpp)
    target_link_libraries(AYSequenceCookTool PRIVATE AYSequenceAssetIO AYAssetFormat)
    if(BUILD_TESTING)
        add_test(NAME AYSequenceCookToolSmoke COMMAND AYSequenceCookTool
            "${CMAKE_CURRENT_SOURCE_DIR}/examples/minimal.seq"
            "${CMAKE_CURRENT_SOURCE_DIR}/examples"
            "${CMAKE_CURRENT_BINARY_DIR}/minimal.sqc")
    endif()
endif()
if(TARGET AYResource AND TARGET AYSequenceAssetIO)
    add_library(AYSequenceResourceIntegration STATIC src/SequenceResource.cpp)
    add_library(AYSequence::Resources ALIAS AYSequenceResourceIntegration)
    target_link_libraries(AYSequenceResourceIntegration PUBLIC AYSequenceAssetIO AYResource
        PRIVATE AYAssetFormat)
endif()
if(TARGET AYResourceImport AND TARGET AYSequenceAssetIO)
    add_library(AYSequenceCookIntegration STATIC src/SequenceConverter.cpp)
    target_link_libraries(AYSequenceCookIntegration PUBLIC
        AYSequenceAssetIO AYResource AYStorage AYAssetFormat)
    target_link_libraries(AYResourceImport PRIVATE AYSequenceCookIntegration)
    target_compile_definitions(AYResourceImport PRIVATE AY_RESOURCE_HAS_SEQUENCE=1)
    target_compile_definitions(AYResource PRIVATE AY_RESOURCE_HAS_SEQUENCE=1)
    if(TARGET AYProjectBuild)
        target_link_libraries(AYProjectBuild PRIVATE AYSequenceAssetIO)
        target_compile_definitions(AYProjectBuild PRIVATE AY_RESOURCE_HAS_SEQUENCE=1)
    endif()
    if(TARGET AYResourceCook)
        target_link_libraries(AYResourceCook PRIVATE AYSequenceAssetIO)
        target_compile_definitions(AYResourceCook PRIVATE AY_RESOURCE_HAS_SEQUENCE=1)
    endif()
    if(TARGET AYResource_Tests)
        target_link_libraries(AYResource_Tests PRIVATE AYSequenceAssetIO)
        target_compile_definitions(AYResource_Tests PRIVATE AY_RESOURCE_HAS_SEQUENCE=1)
    endif()
endif()
if(TARGET AYAudio AND TARGET AYResource)
    add_library(AYSequenceAudioIntegration STATIC src/SequenceAudioPlayer.cpp)
    add_library(AYSequence::AudioIntegration ALIAS AYSequenceAudioIntegration)
    target_link_libraries(AYSequenceAudioIntegration PUBLIC AYSequence AYAudio AYResource)
endif()
if(TARGET AYProgressionCore)
    add_library(AYSequenceDialogueIntegration STATIC src/DialogueSequencePlayback.cpp)
    add_library(AYSequence::DialogueIntegration ALIAS AYSequenceDialogueIntegration)
    target_link_libraries(AYSequenceDialogueIntegration PUBLIC AYSequence AYProgressionCore)
endif()
if(TARGET AYSequenceDialogueIntegration AND TARGET AYAudio AND TARGET AYResource)
    add_library(AYSequenceDialogueAudioIntegration STATIC src/DialogueVoicePlayer.cpp)
    add_library(AYSequence::DialogueAudioIntegration ALIAS AYSequenceDialogueAudioIntegration)
    target_link_libraries(AYSequenceDialogueAudioIntegration PUBLIC
        AYSequenceDialogueIntegration AYAudio AYResource)
endif()
if(TARGET AYUI AND TARGET AYLocalization)
    add_library(AYSequenceUIIntegration STATIC src/SequenceTextPresenter.cpp)
    add_library(AYSequence::UIIntegration ALIAS AYSequenceUIIntegration)
    target_link_libraries(AYSequenceUIIntegration PUBLIC AYSequence AYUI AYLocalization)
endif()
if(TARGET AYSequenceDialogueIntegration AND TARGET AYSequenceUIIntegration)
    add_library(AYSequenceDialogueUIIntegration STATIC src/DialogueTextPresenter.cpp)
    add_library(AYSequence::DialogueUIIntegration ALIAS AYSequenceDialogueUIIntegration)
    target_link_libraries(AYSequenceDialogueUIIntegration PUBLIC
        AYSequenceDialogueIntegration AYSequenceUIIntegration)
endif()
option(AY_SEQUENCE_BUILD_ENTITY_INTEGRATION "Build optional Entity/Host sequence integration" ${AY_BUILD_RUNTIME})
if(AY_SEQUENCE_BUILD_ENTITY_INTEGRATION)
    if(NOT TARGET AYEntityCore)
        message(FATAL_ERROR "AYSequenceEntityIntegration requires AYEntityCore; disable integration for standalone Core")
    endif()
    add_library(AYSequenceEntityIntegration STATIC src/SceneSequenceSession.cpp src/SequenceRuntimeModule.cpp)
    add_library(AYSequence::EntityIntegration ALIAS AYSequenceEntityIntegration)
    target_link_libraries(AYSequenceEntityIntegration PUBLIC AYSequence AYEntityCore AYGameLoop)
endif()
set(_sequence_animation_default OFF)
if(TARGET AYEntityAnimationIntegration AND TARGET AYSequenceEntityIntegration)
    set(_sequence_animation_default ON)
endif()
option(AY_SEQUENCE_BUILD_ANIMATION_INTEGRATION "Build optional skeletal sequence integration" ${_sequence_animation_default})
if(AY_SEQUENCE_BUILD_ANIMATION_INTEGRATION)
    if(NOT TARGET AYEntityAnimationIntegration OR NOT TARGET AYSequenceEntityIntegration)
        message(FATAL_ERROR "Sequence animation integration requires Entity/Animation integrations")
    endif()
    add_library(AYSequenceAnimationIntegration STATIC src/AnimationSequenceDriver.cpp)
    add_library(AYSequence::AnimationIntegration ALIAS AYSequenceAnimationIntegration)
    target_link_libraries(AYSequenceAnimationIntegration PUBLIC AYSequenceEntityIntegration AYEntityAnimationIntegration)
endif()
if(TARGET AYRenderer)
    add_library(AYSequenceRendererIntegration STATIC src/LightShowPresenter.cpp src/LightShowFogPresenter.cpp)
    add_library(AYSequence::RendererIntegration ALIAS AYSequenceRendererIntegration)
    target_link_libraries(AYSequenceRendererIntegration PUBLIC AYSequence AYRenderer)
    if(WIN32 AND TARGET AYSequenceAssetIO)
        add_executable(AYLightShow_Studio demo/LightShowStudio.cpp)
        target_link_libraries(AYLightShow_Studio PRIVATE AYSequenceRendererIntegration AYSequenceAssetIO AYSequenceEditorCore comctl32 comdlg32 user32 gdi32)
        if(TARGET AYAudio)
            target_link_libraries(AYLightShow_Studio PRIVATE AYAudio)
            target_compile_definitions(AYLightShow_Studio PRIVATE AY_LIGHTSHOW_AUDIO=1)
        endif()
        target_compile_definitions(AYLightShow_Studio PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
        ay_stage_production_shaders(AYLightShow_Studio)
    endif()
endif()
if(BUILD_TESTING AND TARGET AYTest AND TARGET AYSequenceAssetIO)
    add_subdirectory(unittest)
endif()
