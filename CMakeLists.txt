cmake_minimum_required(VERSION 3.25)
project(AYSequence VERSION 0.1.0 LANGUAGES CXX)

add_library(AYSequence STATIC src/Sequence.cpp src/LightShow.cpp)
add_library(AYSequence::Core ALIAS AYSequence)
target_compile_features(AYSequence PUBLIC cxx_std_23)
if(NOT TARGET AYAssetFormat)
    add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../AYFoundation/AYAssetFormat
        ${CMAKE_CURRENT_BINARY_DIR}/AYAssetFormat EXCLUDE_FROM_ALL)
endif()
target_link_libraries(AYSequence PRIVATE AYAssetFormat)
target_include_directories(AYSequence PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
)
add_library(AYSequenceEditorCore STATIC src/SequenceAuthoring.cpp)
add_library(AYSequence::EditorCore ALIAS AYSequenceEditorCore)
target_link_libraries(AYSequenceEditorCore PUBLIC AYSequence)

option(AY_SEQUENCE_BUILD_ASSET_IO "Build sequence JSON/file support (requires AYIO)" ${AY_BUILD_RUNTIME})
if(AY_SEQUENCE_BUILD_ASSET_IO)
    find_package(nlohmann_json CONFIG REQUIRED)
    if(NOT TARGET AYIO)
        message(FATAL_ERROR "AYSequenceAssetIO requires AYIO; disable AY_SEQUENCE_BUILD_ASSET_IO for standalone Core")
    endif()
    add_library(AYSequenceAssetIO STATIC src/SequenceAssetIO.cpp src/LightShowAssetIO.cpp)
    add_library(AYSequence::AssetIO ALIAS AYSequenceAssetIO)
    target_link_libraries(AYSequenceAssetIO PUBLIC AYSequence AYIO PRIVATE AYAssetFormat nlohmann_json::nlohmann_json)
endif()
option(AY_SEQUENCE_BUILD_ENTITY_INTEGRATION "Build optional Entity/Host sequence integration" ${AY_BUILD_RUNTIME})
if(AY_SEQUENCE_BUILD_ENTITY_INTEGRATION)
    if(NOT TARGET AYEntityCore)
        message(FATAL_ERROR "AYSequenceEntityIntegration requires AYEntityCore; disable integration for standalone Core")
    endif()
    add_library(AYSequenceEntityIntegration STATIC src/SceneSequenceSession.cpp src/SequenceRuntimeModule.cpp)
    add_library(AYSequence::EntityIntegration ALIAS AYSequenceEntityIntegration)
    target_link_libraries(AYSequenceEntityIntegration PUBLIC AYSequence AYEntityCore AYGameLoop)
endif()
set(_sequence_animation_default OFF)
if(TARGET AYEntityAnimationIntegration AND TARGET AYSequenceEntityIntegration)
    set(_sequence_animation_default ON)
endif()
option(AY_SEQUENCE_BUILD_ANIMATION_INTEGRATION "Build optional skeletal sequence integration" ${_sequence_animation_default})
if(AY_SEQUENCE_BUILD_ANIMATION_INTEGRATION)
    if(NOT TARGET AYEntityAnimationIntegration OR NOT TARGET AYSequenceEntityIntegration)
        message(FATAL_ERROR "Sequence animation integration requires Entity/Animation integrations")
    endif()
    add_library(AYSequenceAnimationIntegration STATIC src/AnimationSequenceDriver.cpp)
    add_library(AYSequence::AnimationIntegration ALIAS AYSequenceAnimationIntegration)
    target_link_libraries(AYSequenceAnimationIntegration PUBLIC AYSequenceEntityIntegration AYEntityAnimationIntegration)
endif()
if(TARGET AYRenderer)
    add_library(AYSequenceRendererIntegration STATIC src/LightShowPresenter.cpp)
    add_library(AYSequence::RendererIntegration ALIAS AYSequenceRendererIntegration)
    target_link_libraries(AYSequenceRendererIntegration PUBLIC AYSequence AYRenderer)
    if(WIN32 AND TARGET AYSequenceAssetIO)
        add_executable(AYLightShow_Studio demo/LightShowStudio.cpp)
        target_link_libraries(AYLightShow_Studio PRIVATE AYSequenceRendererIntegration AYSequenceAssetIO comctl32 comdlg32 user32 gdi32)
        target_compile_definitions(AYLightShow_Studio PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
        ay_stage_production_shaders(AYLightShow_Studio)
    endif()
endif()
if(BUILD_TESTING AND TARGET AYTest AND TARGET AYSequenceAssetIO)
    add_subdirectory(unittest)
endif()
